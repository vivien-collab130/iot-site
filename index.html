/* PROJECT A - Arduino -> Website ("Doorbell") Board: Arduino UNO R4 WiFi (Plug and Make Kit) Modulino: Buttons + Buzzer RATE LIMIT: Adafruit IO free accounts allow 30 messages per minute. This sketch refuses to send more than one message every 2 seconds. Do not remove that check. */ #include #include #include #include "arduino_secrets.h" // ---------- SETTINGS ---------- const char* ssid = SECRET_SSID; const char* pass = SECRET_PASS; const char* ioUser = IO_USERNAME; const char* ioKey = IO_KEY; const char broker[] = "io.adafruit.com"; const int port = 1883; const char feedKey[] = "doorbell"; const char clientId[] = "eagle-doorbell-vivien"; // Minimum time between messages (milliseconds). Keep this >= 2000. const unsigned long MIN_GAP_MS = 2000; // ------------------------------ WiFiClient wifiClient; MqttClient mqttClient(wifiClient); ModulinoButtons buttons; ModulinoBuzzer buzzer; String topic; unsigned long lastSendMs = 0; void setup() { Serial.begin(9600); delay(1500); Modulino.begin(); buttons.begin(); buzzer.begin(); buttons.setLeds(true, true, true); // all three button lights on topic = String(ioUser) + "/feeds/" + feedKey; connectWiFi(); connectMQTT(); Serial.println("Ready. Press A, B, or C."); } void loop() { mqttClient.poll(); if (!mqttClient.connected()) { Serial.println("MQTT disconnected. Reconnecting..."); connectMQTT(); } if (buttons.update()) { if (buttons.isPressed('A')) { sendEvent("A"); } if (buttons.isPressed('B')) { sendEvent("B"); } if (buttons.isPressed('C')) { sendEvent("C"); playHorn(); // HONK HONK on the Buzzer } } } // Car-horn style "HONK HONK" on the Modulino Buzzer. void playHorn() { for (int i = 0; i < 2; i++) { buzzer.tone(400, 450); // low, loud note = horn delay(450 + 150); // wait for the honk, plus a short gap } } // Publishes one message to the feed, but only if enough time has passed. void sendEvent(const char* value) { unsigned long now = millis(); if (now - lastSendMs < MIN_GAP_MS) { Serial.println("Too soon - ignored (rate limit protection)."); return; } lastSendMs = now; Serial.print("Sending: "); Serial.println(value); mqttClient.beginMessage(topic); mqttClient.print(value); mqttClient.endMessage(); } void connectWiFi() { Serial.print("Connecting to WiFi: "); Serial.println(ssid); while (WiFi.begin(ssid, pass) != WL_CONNECTED) { Serial.print("."); delay(3000); } Serial.print("\nConnected. IP address: "); Serial.println(WiFi.localIP()); } void connectMQTT() { mqttClient.setId(clientId); mqttClient.setUsernamePassword(ioUser, ioKey); Serial.print("Connecting to Adafruit IO as "); Serial.println(ioUser); while (!mqttClient.connect(broker, port)) { Serial.print("MQTT connect failed, error code = "); Serial.println(mqttClient.connectError()); Serial.println("Check IO_USERNAME / IO_KEY in arduino_secrets.h. Retrying in 5 s..."); delay(5000); } Serial.println("Connected to Adafruit IO!"); }
